#!/usr/bin/env bash
set -eo pipefail  # Terminate script immediately if any piped command fails

# 1. Environment Setup
REF="/mnt/d/genomes/Pvannamei_2024/GCF_042767895.1_ASM4276789v1_genomic.fna"
RAW_DIR="/mnt/d/CHIP RESULT CONTROL K4 AND K27/N2521626_30-1207159103_A251104055_Delivery_20251105101726/Report/Result/00_Rawdata"
OUT_DIR="/mnt/d/CHIP RESULT CONTROL K4 AND K27/ANNOCATION/BAM_2024"
TMP_DIR="${OUT_DIR}/tmp"

mkdir -p "$OUT_DIR" "$TMP_DIR"

# 2. Priority List
SAMPLES=(K27-I1 K27-W1 K27-I2 K27-W2 K27-I3 K27-W3)

# 3. Execution Loop
for sample in "${SAMPLES[@]}"; do
    echo "============================================"
    echo "PROCESSING: $sample"
    echo "START: $(date)"
    echo "============================================"

    RAW_BAM="${OUT_DIR}/${sample}_raw.bam"
    NAMESORT_BAM="${OUT_DIR}/${sample}_namesort.bam"
    FIXMATE_BAM="${OUT_DIR}/${sample}_fixmate.bam"
    POSSORT_BAM="${OUT_DIR}/${sample}_positionsort.bam"
    DEDUP_BAM="${OUT_DIR}/${sample}_2024_unique_dedup.bam"

    # Step A: Alignment with Read Groups & Quality Filter (MAPQ >= 30)
    rg_header="@RG\tID:${sample}\tSM:${sample}\tPL:ILLUMINA"
    bwa mem -t 6 -R "$rg_header" "$REF" \
        "$RAW_DIR/${sample}_L1_1.fq.gz" \
        "$RAW_DIR/${sample}_L1_2.fq.gz" | \
        samtools view -bhS -q 30 - > "$RAW_BAM"

    # Step B: Name Sort for fixmate
    samtools sort -n -@ 4 -T "$TMP_DIR/${sample}_nsort" -o "$NAMESORT_BAM" "$RAW_BAM"

    # Step C: Add ms and MC tags required for markdup
    samtools fixmate -m "$NAMESORT_BAM" "$FIXMATE_BAM"

    # Step D: Coordinate Sort
    samtools sort -@ 4 -T "$TMP_DIR/${sample}_psort" -o "$POSSORT_BAM" "$FIXMATE_BAM"

    # Step E: Mark and Remove Duplicates (-r removes PCR duplicates)
    samtools markdup -r -@ 4 "$POSSORT_BAM" "$DEDUP_BAM"

    # Step F: Index Deduplicated BAM
    samtools index "$DEDUP_BAM"

    # Step G: Generate IGV TDF
    igvtools count -z 5 -w 25 "$DEDUP_BAM" "${OUT_DIR}/${sample}_2024_unique_dedup.tdf" "$REF"

    # Cleanup Intermediate Files
    rm -f "$RAW_BAM" "$NAMESORT_BAM" "$FIXMATE_BAM" "$POSSORT_BAM"

    echo "SUCCESS: Completed $sample at $(date)"
done
