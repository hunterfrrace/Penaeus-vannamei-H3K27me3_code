## 2. Helper functions: identify annotation and gene-ID columns
## ============================================================

find_column_by_pattern <- function(
  column_names,
  patterns
) {

  for (pattern in patterns) {

    hits <- grep(
      pattern,
      column_names,
      ignore.case = TRUE,
      value = TRUE
    )

    if (length(hits) > 0) {
      return(hits[1])
    }
  }

  NULL
}


extract_ncbi_gene_id <- function(x) {

  x <- as.character(x)

  ## Handles:
  ## LOC113799894
  ## GeneID:113799894
  ## 113799894
  ## "LOC113799894, LOC113799895"

  loc_id <- stringr::str_extract(
    x,
    "(?<=LOC)[0-9]+"
  )

  geneid_id <- stringr::str_extract(
    x,
    "(?<=GeneID:)[0-9]+"
  )

  numeric_id <- stringr::str_extract(
    x,
    "[0-9]+"
  )

  id <- ifelse(
    !is.na(loc_id),
    loc_id,
    ifelse(
      !is.na(geneid_id),
      geneid_id,
      numeric_id
    )
  )

  trimws(id)
}


is_intergenic_annotation <- function(x) {

  x <- tolower(
    trimws(
      as.character(x)
    )
  )

  grepl(
    "distal.*intergenic|intergenic",
    x,
    ignore.case = TRUE
  )
}


is_promoter_annotation <- function(x) {

  x <- tolower(
    trimws(
      as.character(x)
    )
  )

  grepl(
    "promoter|promoter-tss|tss",
    x,
    ignore.case = TRUE
  )
}


is_gene_associated_annotation <- function(x) {

  ## Gene-associated = not intergenic.
  ## This retains promoter, UTR, exon, intron, and downstream categories.

  !is_intergenic_annotation(x)
}


## ============================================================
## 3. Read one full peak-annotation CSV
## ============================================================

read_full_peak_annotation <- function(filepath) {

  message("\n============================================================")
  message("Reading annotation file: ", basename(filepath))
  message("============================================================")

  df <- data.table::fread(
    filepath,
    fill = TRUE,
    data.table = FALSE,
    showProgress = FALSE
  )

  if (nrow(df) == 0) {
    stop(
      paste(
        "Annotation file is empty:",
        filepath
      )
    )
  }

  colnames(df) <- make.names(
    colnames(df),
    unique = TRUE
  )

  cat("\nColumns found:\n")
  print(colnames(df))

  annotation_col <- find_column_by_pattern(
    colnames(df),
    c(
      "^annotation$",
      "annotation",
      "feature",
      "genomic.*region",
      "region",
      "category",
      "location"
    )
  )

  gene_col <- find_column_by_pattern(
    colnames(df),
    c(
      "entrez",
      "^geneid$",
      "gene_id",
      "geneid",
      "gene.*id",
      "nearest.*gene",
      "gene"
    )
  )

  distance_col <- find_column_by_pattern(
    colnames(df),
    c(
      "distance.*tss",
      "dist.*tss",
      "distance"
    )
  )

  cat("\nSelected columns:\n")
  cat("Annotation column: ", annotation_col, "\n", sep = "")
  cat("Gene-ID column: ", gene_col, "\n", sep = "")
  cat("Distance column: ", distance_col, "\n", sep = "")

  if (is.null(annotation_col)) {
    stop(
      paste(
        "Could not detect a genomic annotation column.",
        "\nAvailable columns:",
        paste(colnames(df), collapse = ", ")
      )
    )
  }

  if (is.null(gene_col)) {
    stop(
      paste(
        "Could not detect a gene ID column.",
        "\nAvailable columns:",
        paste(colnames(df), collapse = ", ")
      )
    )
  }

  df$annotation_clean <- trimws(
    as.character(
      df[[annotation_col]]
    )
  )

  df$gene_id_clean <- extract_ncbi_gene_id(
    df[[gene_col]]
  )

  df <- df[
    !is.na(df$gene_id_clean) &
      df$gene_id_clean != "",
    ,
    drop = FALSE
  ]

  if (nrow(df) == 0) {
    stop(
      paste(
        "No usable gene IDs were extracted from:",
        basename(filepath)
      )
    )
  }

  df$intergenic_flag <- is_intergenic_annotation(
    df$annotation_clean
  )

  df$gene_associated_flag <- is_gene_associated_annotation(
    df$annotation_clean
  )

  df$promoter_flag <- is_promoter_annotation(
    df$annotation_clean
  )

  ## If the annotation has no literal promoter label but includes a
  ## TSS-distance column, use |distance| <= 3 kb as a fallback.
  if (
    sum(df$promoter_flag, na.rm = TRUE) == 0 &&
    !is.null(distance_col)
  ) {

    tss_distance <- suppressWarnings(
      as.numeric(
        df[[distance_col]]
      )
    )

    df$promoter_flag <- !is.na(tss_distance) &
      abs(tss_distance) <= 3000

    message(
      "No explicit promoter label found; ",
      "using |distance to TSS| <= 3,000 bp."
    )
  }

  all_peak_associated_genes <- unique(
    df$gene_id_clean
  )

  gene_associated_genes <- unique(
    df$gene_id_clean[
      df$gene_associated_flag
    ]
  )

  promoter_genes <- unique(
    df$gene_id_clean[
      df$promoter_flag
    ]
  )

  intergenic_genes <- unique(
    df$gene_id_clean[
      df$intergenic_flag
    ]
  )

  annotation_counts <- as.data.frame(
    sort(
      table(df$annotation_clean),
      decreasing = TRUE
    )
  )

  colnames(annotation_counts) <- c(
    "Annotation",
    "Peak_rows_with_gene_ID"
  )

  message(
    "Peak rows with usable gene IDs: ",
    nrow(df)
  )

  message(
    "Unique genes from any annotated peak: ",
    length(all_peak_associated_genes)
  )

  message(
    "Unique genes from non-intergenic peaks: ",
    length(gene_associated_genes)
  )

  message(
    "Unique genes from promoter/TSS peaks: ",
    length(promoter_genes)
  )

  message(
    "Unique genes assigned to intergenic peaks: ",
    length(intergenic_genes)
  )

  list(
    all = all_peak_associated_genes,
    gene_associated = gene_associated_genes,
    promoter = promoter_genes,
    intergenic = intergenic_genes,
    data = df,
    annotation_counts = annotation_counts,
    annotation_column = annotation_col,
    gene_column = gene_col,
    distance_column = distance_col
  )
}

## ============================================================
## 8. Run topGO for one ontology
## ============================================================

run_topgo_ont <- function(
  foreground_genes,
  background_genes,
  gene2GO,
  ontology,
  p_cutoff = 0.05
) {

  background_genes <- unique(
    background_genes
  )

  foreground_genes <- intersect(
    unique(foreground_genes),
    background_genes
  )

  if (length(foreground_genes) < 2) {
    message(
      "  ",
      ontology,
      ": fewer than two foreground genes in GO background; skipping."
    )

    return(NULL)
  }

  gene_vector <- as.integer(
    background_genes %in% foreground_genes
  )

  names(gene_vector) <- background_genes

  selected_fun <- function(x) {
    x == 1
  }

  suppressMessages(
    suppressWarnings({

      go_data <- new(
        "topGOdata",
        ontology = ontology,
        allGenes = gene_vector,
        geneSel = selected_fun,
        annot = annFUN.gene2GO,
        gene2GO = gene2GO,
        nodeSize = 1
      )
    })
  )

  fisher_result <- suppressMessages(
    runTest(
      go_data,
      algorithm = "classic",
      statistic = "fisher"
    )
  )

  n_terms <- length(
    score(fisher_result)
  )

  if (n_terms == 0) {
    return(NULL)
  }

  result_table <- GenTable(
    go_data,
    pvalue = fisher_result,
    orderBy = "pvalue",
    topNodes = n_terms
  )

  result_table$pvalue <- suppressWarnings(
    as.numeric(
      gsub(
        "^[< ]+",
        "",
        as.character(
          result_table$pvalue
        )
      )
    )
  )

  result_table$pvalue[
    is.na(result_table$pvalue)
  ] <- 1e-30

  result_table <- result_table[
    !is.na(result_table$pvalue) &
      result_table$pvalue < p_cutoff,
    ,
    drop = FALSE
  ]

  if (nrow(result_table) == 0) {
    return(NULL)
  }

  result_table <- result_table[
    order(result_table$pvalue),
    ,
    drop = FALSE
  ]

  colnames(result_table)[
    colnames(result_table) == "GO.ID"
  ] <- "GO_ID"

  colnames(result_table)[
    colnames(result_table) == "Term"
  ] <- "GO_term"

  colnames(result_table)[
    colnames(result_table) == "Annotated"
  ] <- "bg_in_term"

  colnames(result_table)[
    colnames(result_table) == "Significant"
  ] <- "fg_in_term"

  result_table$fg_in_term <- as.integer(
    result_table$fg_in_term
  )

  result_table$bg_in_term <- as.integer(
    result_table$bg_in_term
  )

  result_table$n_fg <- length(
    foreground_genes
  )

  result_table$n_bg <- length(
    background_genes
  )

  result_table$GeneRatio <- result_table$fg_in_term /
    result_table$n_fg

  result_table$BgRatio <- result_table$bg_in_term /
    result_table$n_bg

  result_table$ontology <- ontology

  result_table$minus_log10_p <- -log10(
    result_table$pvalue
  )

  attr(result_table, "topGOdata") <- go_data
  attr(result_table, "testResult") <- fisher_result

  result_table
}


## ============================================================
## 9. Run BP, MF, and CC
## ============================================================

run_topgo_all <- function(
  foreground_genes,
  background_genes,
  gene2GO,
  p_cutoff = 0.05
) {

  message("  Running BP...")
  bp <- run_topgo_ont(
    foreground_genes,
    background_genes,
    gene2GO,
    ontology = "BP",
    p_cutoff = p_cutoff
  )

  message("  Running MF...")
  mf <- run_topgo_ont(
    foreground_genes,
    background_genes,
    gene2GO,
    ontology = "MF",
    p_cutoff = p_cutoff
  )

  message("  Running CC...")
  cc <- run_topgo_ont(
    foreground_genes,
    background_genes,
    gene2GO,
    ontology = "CC",
    p_cutoff = p_cutoff
  )

  list(
    BP = bp,
    MF = mf,
    CC = cc
  )
}


## ============================================================
## 10. GO bar-plot function
## ============================================================

plot_go_bars <- function(
  go_results,
  title_label,
  top_n = 20,
  out_file = NULL
) {

  ontology_order <- c(
    "BP",
    "MF",
    "CC"
  )

  ontology_names <- c(
    "BP" = "Biological process",
    "MF" = "Molecular function",
    "CC" = "Cellular component"
  )

  ontology_colors <- c(
    "BP" = "#2980B9",
    "MF" = "#C0392B",
    "CC" = "#27AE60"
  )

  plot_dt <- rbindlist(
    lapply(
      ontology_order,
      function(ontology) {

        df <- go_results[[ontology]]

        if (is.null(df) || nrow(df) == 0) {
          return(NULL)
        }

        df <- as.data.table(
          copy(df)
        )

        setorder(
          df,
          pvalue,
          -fg_in_term
        )

        df <- head(
          df,
          top_n
        )

        df[
          ,
          ontology := ontology
        ]

        df[
          ,
          GO_term := stringr::str_trim(
            GO_term
          )
        ]

        df[
          ,
          GO_label := paste0(
            GO_term,
            " [",
            GO_ID,
            "]"
          )
        ]

        df
      }
    ),
    fill = TRUE
  )

  if (nrow(plot_dt) == 0) {
    message(
      "  No significant GO terms available to plot for: ",
      title_label
    )

    return(invisible(NULL))
  }

  plot_dt[
    ,
    Ontology_label := ontology_names[ontology]
  ]

  plot_dt[
    ,
    Ontology_label := factor(
      Ontology_label,
      levels = ontology_names[ontology_order]
    )
  ]

  ## Create a unique ordering label so repeated GO-term names
  ## do not cause factor-level conflicts.
  plot_dt[
    ,
    GO_label_unique := paste0(
      GO_label,
      "___",
      ontology,
      "___",
      seq_len(.N)
    )
  ]

  plot_dt[
    ,
    GO_label_unique := factor(
      GO_label_unique,
      levels = rev(
        unique(
          GO_label_unique[
            order(
              ontology,
              fg_in_term,
              pvalue
            )
          ]
        )
      )
    )
  ]

  p <- ggplot(
    plot_dt,
    aes(
      x = GO_label_unique,
      y = fg_in_term,
      fill = ontology
    )
  ) +

    geom_col(
      width = 0.72
    ) +

    geom_text(
      aes(
        label = fg_in_term
      ),
      hjust = -0.15,
      size = 3
    ) +

    coord_flip() +

    facet_grid(
      Ontology_label ~ .,
      scales = "free_y",
      space = "free_y"
    ) +

    scale_fill_manual(
      values = ontology_colors,
      guide = "none"
    ) +

    scale_x_discrete(
      labels = function(x) {
        sub(
          "___(BP|MF|CC)___[0-9]+$",
          "",
          x
        )
      }
    ) +

    scale_y_continuous(
      expand = expansion(
        mult = c(0, 0.20)
      )
    ) +

    labs(
      title = title_label,
      subtitle = paste0(
        "Gene-associated, non-intergenic H3K27me3 peaks; ",
        "top ",
        top_n,
        " significant terms per ontology; classic Fisher p < 0.05"
      ),
      x = "GO term",
      y = "Foreground gene count"
    ) +

    theme_bw(base_size = 10) +

    theme(
      plot.title = element_text(
        hjust = 0.5,
        face = "bold",
        size = 13
      ),

      plot.subtitle = element_text(
        hjust = 0.5,
        size = 9,
        colour = "grey35"
      ),

      axis.text.y = element_text(
        size = 8
      ),

      axis.text.x = element_text(
        size = 8
      ),

      strip.background = element_rect(
        fill = "grey90",
        colour = "grey60"
      ),

      strip.text.y = element_text(
        face = "bold"
      ),

      panel.grid.major.y = element_blank(),

      panel.grid.minor = element_blank(),

      plot.margin = margin(
        8,
        25,
        8,
        8
      )
    )

  if (!is.null(out_file)) {

    plot_height <- max(
      8,
      min(
        22,
        nrow(plot_dt) * 0.35 + 3
      )
    )

    ggsave(
      filename = out_file,
      plot = p,
      width = 14,
      height = plot_height,
      units = "in",
      dpi = 300,
      bg = "white",
      limitsize = FALSE
    )

    message(
      "  GO bar plot saved: ",
      out_file
    )
  }

  invisible(p)
}


## ============================================================
## 11. Optional GO DAG function
## ============================================================

plot_go_dag <- function(
  go_result_df,
  ontology,
  label,
  top_n = 15,
  out_file
) {

  if (is.null(go_result_df) || nrow(go_result_df) == 0) {
    return(invisible(NULL))
  }

  go_data <- attr(
    go_result_df,
    "topGOdata"
  )

  test_result <- attr(
    go_result_df,
    "testResult"
  )

  if (is.null(go_data) || is.null(test_result)) {
    return(invisible(NULL))
  }

  n_show <- min(
    top_n,
    nrow(go_result_df)
  )

  ontology_name <- c(
    "BP" = "Biological Process",
    "MF" = "Molecular Function",
    "CC" = "Cellular Component"
  )[
    ontology
  ]

  tryCatch(
    {

      pdf(
        out_file,
        width = 16,
        height = 12,
        onefile = TRUE
      )

      suppressMessages(
        showSigOfNodes(
          go_data,
          score(test_result),
          firstSigNodes = n_show,
          useInfo = "all"
        )
      )

      mtext(
        paste0(
          label,
          " — ",
          ontology_name,
          " | top ",
          n_show,
          " terms | classic Fisher p < 0.05"
        ),
        side = 3,
        line = 0.5,
        cex = 0.9,
        font = 2,
        outer = TRUE
      )

      dev.off()

      message(
        "  GO DAG saved: ",
        out_file
      )
    },

    error = function(e) {

      try(
        dev.off(),
        silent = TRUE
      )

      message(
        "  GO DAG failed for ",
        label,
        " / ",
        ontology,
        ": ",
        conditionMessage(e)
      )
    }
  )

  invisible(NULL)
}


## ============================================================
## 12. Master function: run one foreground GO analysis
## ============================================================

run_and_save_go <- function(
  foreground_genes,
  background_genes,
  gene2GO,
  label,
  top_n_bar = 20,
  top_n_dag = 15,
  p_cutoff = 0.05
) {

  foreground_genes <- intersect(
    unique(foreground_genes),
    unique(background_genes)
  )

  foreground_go <- intersect(
    foreground_genes,
    names(gene2GO)
  )

  background_go_current <- intersect(
    unique(background_genes),
    names(gene2GO)
  )

  message("\n════════════════════════════════════════════════════")
  message("ANALYSIS: ", label)
  message("Foreground gene-associated genes: ", length(foreground_genes))
  message("Foreground genes with GO annotation: ", length(foreground_go))
  message("Background genes with GO annotation: ", length(background_go_current))
  message("════════════════════════════════════════════════════")

  if (length(foreground_go) < 2) {

    warning(
      paste(
        "Skipping ",
        label,
        ": fewer than two foreground genes have GO annotation.",
        sep = ""
      )
    )

    return(NULL)
  }

  fwrite(
    data.table(
      GeneID = foreground_go
    ),
    file.path(
      out_dir,
      paste0(
        label,
        "_Foreground_GO_Annotated_Genes.csv"
      )
    )
  )

  go_results <- run_topgo_all(
    foreground_genes = foreground_go,
    background_genes = background_go_current,
    gene2GO = gene2GO,
    p_cutoff = p_cutoff
  )

  significant_summary <- data.table(
    Analysis = label,
    Ontology = c(
      "BP",
      "MF",
      "CC"
    ),
    Significant_terms = vapply(
      go_results,
      function(x) {
        if (is.null(x)) {
          0L
        } else {
          nrow(x)
        }
      },
      integer(1)
    ),
    Foreground_GO_genes = length(foreground_go),
    Background_GO_genes = length(background_go_current)
  )

  fwrite(
    significant_summary,
    file.path(
      out_dir,
      paste0(
        label,
        "_GO_Significant_Term_Summary.csv"
      )
    )
  )

  cat("\nSignificant GO-term counts:\n")
  print(significant_summary)

  barplot_file <- file.path(
    out_dir,
    paste0(
      label,
      "_GO_Barplot.pdf"
    )
  )

  plot_go_bars(
    go_results = go_results,
    title_label = label,
    top_n = top_n_bar,
    out_file = barplot_file
  )

  for (ontology in c("BP", "MF", "CC")) {

    if (
      !is.null(go_results[[ontology]]) &&
      nrow(go_results[[ontology]]) > 0
    ) {

      dag_file <- file.path(
        out_dir,
        paste0(
          label,
          "_GO_DAG_",
          ontology,
          ".pdf"
        )
      )

      plot_go_dag(
        go_result_df = go_results[[ontology]],
        ontology = ontology,
        label = label,
        top_n = top_n_dag,
        out_file = dag_file
      )
    }
  }

  full_results <- rbindlist(
    lapply(
      go_results,
      function(df) {

        if (is.null(df) || nrow(df) == 0) {
          return(NULL)
        }

        output_df <- as.data.table(
          copy(df)
        )

        attr(
          output_df,
          "topGOdata"
        ) <- NULL

        attr(
          output_df,
          "testResult"
        ) <- NULL

        output_df
      }
    ),
    fill = TRUE
  )

  if (nrow(full_results) > 0) {

    fwrite(
      full_results,
      file.path(
        out_dir,
        paste0(
          label,
          "_GO_Results_All_Ontologies.csv"
        )
      )
    )

    write.table(
      full_results,
      file.path(
        out_dir,
        paste0(
          label,
          "_GO_Results_All_Ontologies.txt"
        )
      ),
      sep = "\t",
      row.names = FALSE,
      quote = FALSE
    )
  }

  go_results
}
