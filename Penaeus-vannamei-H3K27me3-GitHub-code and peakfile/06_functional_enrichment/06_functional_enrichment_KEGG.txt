find_column_by_pattern <- function(
        column_names,
        patterns
) {
    
    for (pattern in patterns) {
        
        hits <- grep(
            pattern,
            column_names,
            ignore.case = TRUE,
            value = TRUE
        )
        
        if (length(hits) > 0) {
            return(hits[1])
        }
    }
    
    return(NULL)
}


extract_ncbi_gene_id <- function(x) {
    
    x <- as.character(x)
    
    loc_id <- stringr::str_extract(
        x,
        "(?<=LOC)[0-9]+"
    )
    
    geneid_id <- stringr::str_extract(
        x,
        "(?<=GeneID:)[0-9]+"
    )
    
    numeric_id <- stringr::str_extract(
        x,
        "[0-9]+"
    )
    
    gene_id <- ifelse(
        !is.na(loc_id),
        loc_id,
        ifelse(
            !is.na(geneid_id),
            geneid_id,
            numeric_id
        )
    )
    
    trimws(gene_id)
}


is_intergenic_annotation <- function(x) {
    
    x <- tolower(
        trimws(
            as.character(x)
        )
    )
    
    grepl(
        "distal.*intergenic|intergenic",
        x,
        ignore.case = TRUE
    )
}


is_gene_associated_annotation <- function(x) {
    
    !is_intergenic_annotation(x)
}


# ============================================================
# HELPER: FORMAT Q-VALUE AS MATHEMATICAL EXPRESSION
# ============================================================

format_q_math <- function(q) {
    
    if (is.na(q) || q <= 0) {
        return("q == NA")
    }
    
    exponent <- floor(log10(q))
    mantissa <- q / (10^exponent)
    
    sprintf("q == %.2f %%*%% 10^{%d}", mantissa, exponent)
}

read_annotation_file <- function(
        filepath,
        sample_name
) {
    
    message("\n============================================================")
    message("Reading: ", basename(filepath))
    message("Sample: ", sample_name)
    message("============================================================")
    
    df <- data.table::fread(
        filepath,
        fill = TRUE,
        data.table = FALSE,
        showProgress = FALSE
    )
    
    if (nrow(df) == 0) {
        stop(
            "Annotation file is empty: ",
            filepath
        )
    }
    
    colnames(df) <- make.names(
        colnames(df),
        unique = TRUE
    )
    
    annotation_col <- find_column_by_pattern(
        colnames(df),
        c(
            "^annotation$",
            "annotation",
            "feature",
            "genomic.*region",
            "region",
            "category",
            "location"
        )
    )
    
    gene_col <- find_column_by_pattern(
        colnames(df),
        c(
            "entrez",
            "^geneid$",
            "gene_id",
            "geneid",
            "gene.*id",
            "nearest.*gene",
            "gene"
        )
    )
    
    if (is.null(annotation_col)) {
        stop(
            "Could not detect annotation column in: ",
            basename(filepath)
        )
    }
    
    if (is.null(gene_col)) {
        stop(
            "Could not detect gene-ID column in: ",
            basename(filepath)
        )
    }
    
    df$annotation_clean <- trimws(
        as.character(df[[annotation_col]])
    )
    
    df$gene_id_clean <- extract_ncbi_gene_id(
        df[[gene_col]]
    )
    
    df <- df[
        !is.na(df$gene_id_clean) &
            df$gene_id_clean != "",
        ,
        drop = FALSE
    ]
    
    if (nrow(df) == 0) {
        stop(
            "No usable gene IDs were extracted from: ",
            basename(filepath)
        )
    }
    
    df$intergenic_flag <- is_intergenic_annotation(
        df$annotation_clean
    )
    
    df$gene_associated_flag <- is_gene_associated_annotation(
        df$annotation_clean
    )
    
    all_genes <- unique(
        as.character(df$gene_id_clean)
    )
    
    gene_associated_genes <- unique(
        as.character(
            df$gene_id_clean[
                df$gene_associated_flag
            ]
        )
    )
    
    message(
        "Peak rows with usable gene IDs: ",
        nrow(df)
    )
    
    message(
        "Unique genes from all peaks: ",
        length(all_genes)
    )
    
    message(
        "Unique gene-associated non-intergenic genes: ",
        length(gene_associated_genes)
    )
    
    return(
        list(
            all = all_genes,
            gene_associated = gene_associated_genes,
            data = df
        )
    )
}

get_kegg_category <- function(pathway_id) {
    
    pathway_id <- as.character(pathway_id)
    
    if (grepl("^map0[0-1]", pathway_id)) {
        return("Metabolism")
    }
    
    if (grepl("^map02", pathway_id)) {
        return("Environmental Information Processing")
    }
    
    if (grepl("^map03", pathway_id)) {
        return("Genetic Information Processing")
    }
    
    if (grepl("^map04", pathway_id)) {
        return("Cellular Processes")
    }
    
    if (grepl("^map05", pathway_id)) {
        return("Human Diseases")
    }
    
    if (grepl("^map09", pathway_id)) {
        return("Organismal Systems")
    }
    
    return("Other")
}

# 15. KEGG ENRICHMENT
# ============================================================

run_kegg_enrichment_map_only <- function(
        fg_genes,
        bg_genes,
        kegg_dat
) {
    
    gene2pathway <- kegg_dat$gene2pathway
    pathway2gene <- kegg_dat$pathway2gene
    pathway_info <- kegg_dat$pathway_info
    
    fg_genes <- unique(
        as.character(fg_genes)
    )
    
    bg_genes <- unique(
        as.character(bg_genes)
    )
    
    fg_genes <- intersect(
        fg_genes,
        bg_genes
    )
    
    bg_mapped <- intersect(
        bg_genes,
        names(gene2pathway)
    )
    
    fg_mapped <- intersect(
        fg_genes,
        names(gene2pathway)
    )
    
    n_bg <- length(bg_mapped)
    n_fg <- length(fg_mapped)
    
    message(
        "  Foreground genes mapped to KEGG: ",
        n_fg,
        "/",
        length(fg_genes)
    )
    
    message(
        "  Background genes mapped to KEGG: ",
        n_bg,
        "/",
        length(bg_genes)
    )
    
    if (n_fg < 1 || n_bg < 1) {
        return(NULL)
    }
    
    all_pathways <- unique(
        unlist(
            gene2pathway[bg_mapped],
            use.names = FALSE
        )
    )
    
    all_pathways <- all_pathways[
        grepl(
            "^map[0-9]{5}$",
            all_pathways
        )
    ]
    
    if (length(all_pathways) == 0) {
        return(NULL)
    }
    
    result_list <- lapply(
        all_pathways,
        function(pathway_id) {
            
            pathway_genes <- pathway2gene[[pathway_id]]
            
            if (
                is.null(pathway_genes) ||
                length(pathway_genes) == 0
            ) {
                return(NULL)
            }
            
            fg_pathway_genes <- intersect(
                fg_mapped,
                pathway_genes
            )
            
            bg_pathway_genes <- intersect(
                bg_mapped,
                pathway_genes
            )
            
            a <- length(fg_pathway_genes)
            
            if (a == 0) {
                return(NULL)
            }
            
            b <- n_fg - a
            c_value <- length(bg_pathway_genes) - a
            d <- n_bg - length(bg_pathway_genes)
            
            if (any(c(a, b, c_value, d) < 0)) {
                return(NULL)
            }
            
            fisher_matrix <- matrix(
                c(a, b, c_value, d),
                nrow = 2,
                byrow = TRUE
            )
            
            fisher_result <- tryCatch(
                fisher.test(
                    fisher_matrix,
                    alternative = "greater"
                ),
                error = function(e) {
                    NULL
                }
            )
            
            if (is.null(fisher_result)) {
                return(NULL)
            }
            
            pathway_name <- pathway_info$pathway_name[
                match(
                    pathway_id,
                    pathway_info$pathway_id
                )
            ]
            
            if (
                length(pathway_name) == 0 ||
                is.na(pathway_name) ||
                pathway_name == ""
            ) {
                pathway_name <- pathway_id
            }
            
            odds_ratio <- unname(
                as.numeric(
                    fisher_result$estimate
                )
            )
            
            if (
                is.na(odds_ratio) ||
                is.nan(odds_ratio)
            ) {
                odds_ratio <- Inf
            }
            
            data.frame(
                pathway_id = pathway_id,
                pathway_name = pathway_name,
                fg_in_path = a,
                bg_in_path = length(bg_pathway_genes),
                n_fg = n_fg,
                n_bg = n_bg,
                GeneRatio = a / n_fg,
                BgRatio = length(bg_pathway_genes) / n_bg,
                RichFactor = a / length(bg_pathway_genes),
                pvalue = fisher_result$p.value,
                odds_ratio = odds_ratio,
                stringsAsFactors = FALSE
            )
        }
    )
    
    result_list <- Filter(
        Negate(is.null),
        result_list
    )
    
    if (length(result_list) == 0) {
        return(NULL)
    }
    
    results <- do.call(
        rbind,
        result_list
    )
    
    results$qvalue <- p.adjust(
        results$pvalue,
        method = "BH"
    )
    
    results <- results[
        order(
            results$qvalue,
            results$pvalue,
            -results$fg_in_path
        ),
        ,
        drop = FALSE
    ]
    
    results <- results[
        !duplicated(results$pathway_id),
        ,
        drop = FALSE
    ]
    
    results <- results[
        !duplicated(
            tolower(
                trimws(
                    results$pathway_name
                )
            )
        ),
        ,
        drop = FALSE
    ]
    
    rownames(results) <- NULL
    
    return(results)
}


# ============================================================
# 16. STRICT FILTER
# ============================================================

filter_kegg_results_strict <- function(
        results
) {
    
    if (
        is.null(results) ||
        nrow(results) == 0
    ) {
        return(results)
    }
    
    filtered_results <- results[
        !is.na(results$qvalue) &
            results$qvalue <= q_cutoff &
            !is.na(results$fg_in_path) &
            results$fg_in_path >= min_fg_genes &
            !is.na(results$odds_ratio) &
            (
                is.infinite(results$odds_ratio) |
                    results$odds_ratio >= min_odds_ratio
            ) &
            !is.na(results$RichFactor) &
            is.finite(results$RichFactor) &
            results$RichFactor >= min_rich_factor &
            results$RichFactor < max_rich_factor,
        ,
        drop = FALSE
    ]
    
    filtered_results <- filtered_results[
        order(
            filtered_results$qvalue,
            -filtered_results$fg_in_path,
            -filtered_results$RichFactor
        ),
        ,
        drop = FALSE
    ]
    
    rownames(filtered_results) <- NULL
    
    return(filtered_results)
}


# ============================================================
# 17. PREPARE PLOT DATA (NO WRAPPING, SINGLE-LINE LABELS)
# ============================================================

prepare_plot_data <- function(
        results,
        top_n,
        plot_type = "bar"
) {
    
    if (
        is.null(results) ||
        nrow(results) == 0
    ) {
        return(NULL)
    }
    
    results <- results[
        order(
            results$qvalue,
            -results$fg_in_path
        ),
        ,
        drop = FALSE
    ]
    
    results <- head(
        results,
        top_n
    )
    
    results$kegg_category <- vapply(
        results$pathway_id,
        get_kegg_category,
        character(1)
    )
    
    results$plot_label <- trimws(
        as.character(
            results$pathway_name
        )
    )
    
    results$plot_label <- factor(
        results$plot_label,
        levels = rev(
            unique(
                results$plot_label
            )
        )
    )
    
    return(results)
}

# ============================================================
# 20. MASTER KEGG ANALYSIS FUNCTION
# ============================================================

run_kegg_analysis <- function(
        fg_genes,
        bg_genes,
        label
) {
    
    message("\n============================================================")
    message("KEGG ANALYSIS: ", label)
    message("q-value cutoff: ", q_cutoff)
    message("============================================================")
    
    raw_results <- run_kegg_enrichment_map_only(
        fg_genes = fg_genes,
        bg_genes = bg_genes,
        kegg_dat = kegg_data
    )
    
    if (
        is.null(raw_results) ||
        nrow(raw_results) == 0
    ) {
        message(
            "No raw KEGG results generated for: ",
            label
        )
        return(NULL)
    }
    
    raw_results_file <- file.path(
        out_dir_kegg,
        paste0(
            label,
            "_KEGG_ALL_MAP_ONLY_results.csv"
        )
    )
    
    fwrite(
        as.data.table(raw_results),
        raw_results_file
    )
    
    strict_results <- filter_kegg_results_strict(
        raw_results
    )
    
    strict_results_file <- file.path(
        out_dir_kegg,
        paste0(
            label,
            "_KEGG_STRICT_FILTERED_q0.05_results.csv"
        )
    )
    
    fwrite(
        as.data.table(strict_results),
        strict_results_file
    )
    
    message(
        "Raw MAP pathways: ",
        nrow(raw_results)
    )
    
    message(
        "Strict pathways retained: ",
        nrow(strict_results)
    )
    
    if (
        is.null(strict_results) ||
        nrow(strict_results) == 0
    ) {
        message(
            "No pathways passed the strict filters for: ",
            label
        )
        
        return(
            list(
                raw = raw_results,
                strict = strict_results
            )
        )
    }
    
    bar_file <- file.path(
        out_dir_kegg,
        paste0(
            label,
            "_KEGG_Top30_Barplot_q0.05.png"
        )
    )
    
    bubble_file <- file.path(
        out_dir_kegg,
        paste0(
            label,
            "_KEGG_Top30_Bubbleplot_q0.05.png"
        )
    )
    
    plot_kegg_bar(
        results = strict_results,
        label = label,
        out_file = bar_file
    )
    
    plot_kegg_bubble(
        results = strict_results,
        label = label,
        out_file = bubble_file
    )
    
    return(
        list(
            raw = raw_results,
            strict = strict_results
        )
    )
}

        out_dir_kegg,
        paste0(
            label,
            "_KEGG_Supplementary_Strict_Pathways_q0.05.txt"
        )

        row.names = FALSE,
        quote = FALSE
