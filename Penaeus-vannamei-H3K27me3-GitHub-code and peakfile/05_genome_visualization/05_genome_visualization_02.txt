# =============================================================================# PANEL B - Calculate TSS profile
promoters <- ChIPseeker::getPromoters(TxDb = txdb, upstream = 10000, downstream = 10000)

calculate_tss_profile <- function(peaks_gr, sample_name) {
    tag_matrix <- ChIPseeker::getTagMatrix(peaks_gr, windows = promoters)
    
    # It's a matrix, not a list!
    n_pos <- ncol(tag_matrix)
    x_positions <- seq(-10000, 10000, length.out = n_pos)
    mean_coverage <- colMeans(tag_matrix, na.rm = TRUE)
    lower_ci <- apply(tag_matrix, 2, function(col) quantile(col, 0.025, na.rm = TRUE))
    upper_ci <- apply(tag_matrix, 2, function(col) quantile(col, 0.975, na.rm = TRUE))
    
    data.table(
        Position = x_positions,
        Mean_Coverage = mean_coverage,
        Lower = as.numeric(lower_ci),
        Upper = as.numeric(upper_ci),
        Sample = sample_name
    )
}

panel_b_data <- rbindlist(
    lapply(names(K27_PEAKS), function(sample) {
        calculate_tss_profile(K27_PEAKS[[sample]], sample)
    })
)

annotate_peaks <- function(peaks_gr, sample_name) {
    peak_annotation <- ChIPseeker::annotatePeak(peak = peaks_gr, tssRegion = c(-3000, 3000), TxDb = txdb,
                                               level = "transcript", assignGenomicAnnotation = TRUE, verbose = FALSE)
    annotation_dt <- as.data.table(as.data.frame(peak_annotation))
    annotation_dt[, simple_annotation := {
        ann <- as.character(annotation)
        ifelse(grepl("Promoter \\(<=1kb\\)", ann), "Promoter (<=1kb)",
        ifelse(grepl("Promoter \\(1-2kb\\)", ann), "Promoter (1-2kb)",
        ifelse(grepl("Promoter \\(2-3kb\\)", ann), "Promoter (2-3kb)",
        ifelse(grepl("5' UTR", ann), "5' UTR",
        ifelse(grepl("3' UTR", ann), "3' UTR",
        ifelse(grepl("1st Exon", ann), "1st Exon",
        ifelse(grepl("Exon", ann), "Other Exon",
        ifelse(grepl("1st Intron", ann), "1st Intron",
        ifelse(grepl("Intron", ann), "Other Intron",
        ifelse(grepl("Downstream", ann), "Downstream (<=300bp)",
        ifelse(grepl("Distal Intergenic", ann), "Distal Intergenic", "Other")))))))))))
    }]
    annotation_dt[, Sample := sample_name]
    annotation_dt
}

    complete

panel_c_data <- rbindlist(lapply(names(annotation_list), function(sample) {
    dt <- annotation_list[[sample]][!is.na(distanceToTSS)]
    dt[, side := fifelse(distanceToTSS < 0, "5prime", "3prime")]
    dt[, absolute_distance := abs(distanceToTSS)]
    dt[, bin_label := cut(absolute_distance, breaks = c(-Inf, 1000, 3000, 5000, 10000, 100000, Inf),
                         labels = c("0 bp-1 kb", "1-3 kb", "3-5 kb", "5-10 kb", "10-100 kb", ">100 kb"), include.lowest = TRUE)]
    side_totals <- dt[, .(side_total = .N), by = side]
    counts <- dt[, .N, by = .(side, bin_label)]
    counts <- merge(counts, side_totals, by = "side", all.x = TRUE)
    counts[, pct := 100 * N / side_total]
    counts[, signed_pct := fifelse(side == "5prime", -pct, pct)]
    counts[, Sample := sample]
    counts[, bin_label := factor(bin_label, levels = bin_levels)]
    counts
}))
