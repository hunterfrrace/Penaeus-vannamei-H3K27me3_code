# =============================================================================
# 6. Helper function: fixed-width genomic bins
# =============================================================================

make_manual_bins <- function(
        chrom_summary,
        bin_size = BIN_SIZE_BP
) {
    
    bin_list <- vector(
        mode = "list",
        length = nrow(chrom_summary)
    )
    
    for (i in seq_len(nrow(chrom_summary))) {
        
        chromosome <- chrom_summary$chrom[i]
        
        chromosome_end <- as.integer(
            ceiling(chrom_summary$max_pos[i])
        )
        
        if (
            length(chromosome_end) == 0L ||
            is.na(chromosome_end) ||
            chromosome_end < 1L
        ) {
            next
        }
        
        bin_starts <- seq.int(
            from = 1L,
            to = chromosome_end,
            by = bin_size
        )
        
        bin_ends <- pmin(
            bin_starts + bin_size - 1L,
            chromosome_end
        )
        
        bin_list[[i]] <- GRanges(
            seqnames = chromosome,
            ranges = IRanges(
                start = bin_starts,
                end = bin_ends
            ),
            strand = "*"
        )
    }
    
    bin_list <- bin_list[
        !vapply(
            bin_list,
            is.null,
            logical(1)
        )
    ]
    
    if (length(bin_list) == 0L) {
        stop("No valid genomic bins were created.")
    }
    
    do.call(c, bin_list)
}


# =============================================================================
# 7. Helper function: extract MACS3 signalValue
# =============================================================================

extract_enrichment <- function(peaks_gr) {
    
    metadata_columns <- colnames(mcols(peaks_gr))
    
    preferred_columns <- c(
        "signalValue",
        "signalvalue",
        "fold_enrichment",
        "foldEnrichment",
        "enrichment"
    )
    
    preferred_available <- preferred_columns[
        preferred_columns %in% metadata_columns
    ]
    
    if (length(preferred_available) > 0L) {
        
        selected_column <- preferred_available[1]
        
    } else {
        
        candidate_columns <- grep(
            "signal|fold|enrich",
            metadata_columns,
            ignore.case = TRUE,
            value = TRUE
        )
        
        if (length(candidate_columns) == 0L) {
            stop(
                "Could not identify a MACS3 signalValue/fold-enrichment ",
                "metadata column."
            )
        }
        
        selected_column <- candidate_columns[1]
    }
    
    values <- suppressWarnings(
        as.numeric(mcols(peaks_gr)[[selected_column]])
    )
    
    values <- values[
        is.finite(values) &
            values >= 0
    ]
    
    if (length(values) == 0L) {
        stop("No valid, finite non-negative enrichment values were found.")
    }
    
    list(
        values = values,
        column_name = selected_column
    )
}


# =============================================================================
# 8. Function: make Figure 3 for one replicate
# =============================================================================

make_figure3 <- function(
        peaks_gr,
        sample_id,
        top_n_scaffolds = TOP_N_SCAFFOLDS,
        bin_size = BIN_SIZE_BP
) {
    
    n_peaks <- length(peaks_gr)
    
    panel_a_png_file <- file.path(
        out_dir,
        paste0(
            "Figure3_",
            sample_id,
            "_PanelA_Top15_H3K27me3_Coverage_60pt.png"

    )
    
    panel_a_pdf_file <- file.path(
        out_dir,
        paste0(
            "Figure3_",
            sample_id,
            "_PanelA_Top15_H3K27me3_Coverage_60pt.pdf"
        )
    )
    
    combined_png_file <- file.path(
        out_dir,
        paste0(
            "Figure3_",
            sample_id,
            "_Panel_ABC_Top15_H3K27me3_Coverage_60pt.png"

    combined_pdf_file <- file.path(
        out_dir,
        paste0(
            "Figure3_",
            sample_id,

    message(
        "\n>>> Creating Figure 3 for ",
        sample_id,
        " using ",
        scales::comma(n_peaks),
        " peaks."
    )
    
    # =========================================================================
    # Panel A data: rank scaffolds by total broad-peak coverage
    # =========================================================================
    
    metadata_columns <- colnames(mcols(peaks_gr))
    
    if ("signalValue" %in% metadata_columns) {
        peak_signal <- suppressWarnings(
            as.numeric(peaks_gr$signalValue)
        )
    } else {
        peak_signal <- rep(
            NA_real_,
            length(peaks_gr)
        )
    }
    
    peak_dt <- data.table(
        chrom = as.character(seqnames(peaks_gr)),
        start = start(peaks_gr),
        end = end(peaks_gr),
        peak_width_bp = as.numeric(width(peaks_gr)),
        signal_value = peak_signal
    )
    
    # This table contains all scaffolds with at least one filtered peak.
    scaffold_ranking_all <- peak_dt[
        ,
        .(
            observed_scaffold_length_bp = max(end, na.rm = TRUE),
            peak_count = .N,
            total_peak_coverage_bp = sum(
                peak_width_bp,
                na.rm = TRUE
            ),
            percent_observed_scaffold_covered = round(
                100 *
                    sum(peak_width_bp, na.rm = TRUE) /
                    max(end, na.rm = TRUE),
                3
            ),
            total_signal_value = sum(
                signal_value,
                na.rm = TRUE
            ),
            median_signal_value = median(
                signal_value,
                na.rm = TRUE
            )
        ),
        by = chrom
    ]
    
    scaffold_ranking_all[
        !is.finite(total_signal_value),
        total_signal_value := NA_real_
    ]
    
    scaffold_ranking_all[
        !is.finite(median_signal_value),
        median_signal_value := NA_real_
    ]
    
    # Transparent selection criteria:
    # 1. greatest total broad-peak coverage
    # 2. most called broad peaks
    # 3. longest observed scaffold coordinate
    setorder(
        scaffold_ranking_all,
        -total_peak_coverage_bp,
        -peak_count,
        -observed_scaffold_length_bp,
        chrom
    )
    
    scaffold_ranking_all[
        ,
        Coverage_rank := seq_len(.N)
    ]
    
    scaffold_ranking_selected <- head(
        scaffold_ranking_all,
        top_n_scaffolds
    )
    
    if (nrow(scaffold_ranking_selected) == 0L) {
        stop(
            "No scaffolds with peaks were available for ",
            sample_id,
            "."
        )
    }
    
    chrom_order <- scaffold_ranking_selected$chrom
    
    peaks_selected <- peaks_gr[
        as.character(seqnames(peaks_gr)) %in% chrom_order
    ]
    
    bins_input <- scaffold_ranking_selected[
        ,
        .(
            chrom = chrom,
            max_pos = observed_scaffold_length_bp
        )
    ]
    
    bins_gr <- make_manual_bins(
        chrom_summary = bins_input,
        bin_size = bin_size
    )
    
    bins_gr$peak_count <- countOverlaps(
        query = bins_gr,
        subject = peaks_selected,
        ignore.strand = TRUE
    )
    
    binned_data <- data.table(
        chrom = as.character(seqnames(bins_gr)),
        start = start(bins_gr),
        end = end(bins_gr),
        peak_count = bins_gr$peak_count
    )
    
    binned_data[
        ,
        position_mb := (start + end) / 2e6
    ]
    
    # Highest-ranked scaffold appears at top.
    binned_data[
        ,
        chrom := factor(
            chrom,
            levels = rev(chrom_order)
        )
    ]
    
    # =========================================================================
    # Panel A plot: density tracks
    # =========================================================================
    
    p_a <- ggplot(
        binned_data,
        aes(
            x = position_mb,
            y = peak_count,
            group = chrom
        )
    ) +
        geom_area(
            fill = TRACK_FILL,
            alpha = 0.68,
            color = TRACK_BORDER,
            linewidth = 0.40
        ) +
        facet_grid(
            chrom ~ .,
            switch = "y",
            scales = "free_y",
            space = "free_y",
            axes = "all_y",
            axis.labels = "all_y"
        ) +
        scale_x_continuous(
            labels = function(x) {
                paste0(
                    formatC(
                        x,
                        format = "f",
                        digits = 1
                    ),
                    " Mb"
                )
            },
            expand = expansion(mult = c(0.01, 0.01))
        ) +
        scale_y_continuous(
            n.breaks = 3,
            expand = expansion(mult = c(0, 0.06))
        ) +
        labs(
            title = paste0(
                "a  H3K27me3 density on top scaffolds by broad-peak coverage (",
                sample_id,
                ")"
            ),
            x = "Position on chromosome/scaffold",
            y = "Peaks per 100 kb"
        ) +
        theme_light(
            base_size = TICK_FONT_SIZE,
            base_family = FONT_FAMILY
        ) +
        theme(
            text = element_text(
                family = FONT_FAMILY,
                color = "black"
            ),
            
            plot.title = element_text(
                hjust = 0.5,
                size = PLOT_TITLE_SIZE,
                face = "bold",
                family = FONT_FAMILY,
                color = "black",
                margin = margin(b = 28)
            ),
            
            axis.title.x = element_text(
                size = AXIS_TITLE_SIZE,
                face = "bold",
                family = FONT_FAMILY,
                color = "black",
                margin = margin(t = 20)
            ),
            
            axis.title.y = element_text(
                size = AXIS_TITLE_SIZE,
                face = "bold",
                family = FONT_FAMILY,
                color = "black",
                margin = margin(r = 22)
            ),
            
            axis.text.x = element_text(
                size = TICK_FONT_SIZE,
                family = FONT_FAMILY,
                color = "black",
                margin = margin(t = 8)
            ),
            
            axis.text.y = element_text(
                size = TICK_FONT_SIZE,
                family = FONT_FAMILY,
                color = "black",
                margin = margin(r = 3),
                hjust = 1
            ),
            
            axis.ticks.y = element_line(
                color = "black",
                linewidth = 0.80
            ),
            
            axis.ticks.length.y = grid::unit(
                2,
                "pt"
            ),
            
            axis.ticks.x = element_line(
                color = "black",
                linewidth = 0.80
            ),
            
            axis.ticks.length.x = grid::unit(
                3,
                "pt"
            ),
            
            strip.placement = "outside",
            
            strip.text.y.left = element_text(
                angle = 0,
                size = STRIP_FONT_SIZE,
                family = FONT_FAMILY,
                face = "bold",
                color = "black",
                margin = margin(
                    l = 12,
                    r = 12,
                    unit = "pt"

                color = "#D9D9D9",

            panel.grid.major.y = element_line(
                color = "#EDEDED",
                linewidth = 0.28

            panel.grid.minor = element_blank(),

                t = 28,
                r = 25,

                unit = "pt"

    # =========================================================================

    # =========================================================================
    
    ggsave(
        filename = panel_a_png_file,

        height = PANEL_A_HEIGHT_IN,
        units = "in",

        bg = "white",
        limitsize = FALSE

        plot = p_a,
        width = PANEL_A_WIDTH_IN,
        height = PANEL_A_HEIGHT_IN,
        units = "in",
        device = cairo_pdf,
        bg = "white",
        limitsize = FALSE

            labels = scales::comma,

            axis.title.x = element_text(
                size = AXIS_TITLE_SIZE,

            axis.title.y = element_text(
                size = AXIS_TITLE_SIZE,
                margin = margin(r = 14)

            axis.text.y = element_text(
                size = TICK_FONT_SIZE,
                family = FONT_FAMILY,
                color = "black",
                margin = margin(r = 3),
                hjust = 1

            axis.text.x = element_text(
                size = TICK_FONT_SIZE,
                family = FONT_FAMILY,
                color = "black",
                margin = margin(t = 8)
            ),
            
            axis.ticks.y = element_line(
                color = "black",
                linewidth = 0.80
            ),
            
            axis.ticks.length.y = grid::unit(
                2,
                "pt"
            ),
            
            axis.ticks.x = element_line(
                color = "black",
                linewidth = 0.80
            ),
            
            axis.ticks.length.x = grid::unit(
                3,
                "pt"

            panel.grid.major.y = element_line(
                color = "grey85",

            panel.grid.major.x = element_blank(),

                l = 25,

    # =========================================================================
