############################
# 3. Define exactly four inputs
############################

peak_files <- c(

missing_files <- peak_files[

############################
# 4. Load TxDb
############################

message("Loading custom TxDb...")

txdb <- AnnotationDbi::loadDb(

)

txdb_seqlevels <- GenomeInfoDb::seqlevels(txdb)

message(
    "Number of TxDb sequence levels: ",

)
############################
# 5. Import peak files
############################

message("Importing peak files...")

peak_granges <- list()

for (sample_name in names(peak_files)) {
    
    current_file <- peak_files[[sample_name]]
    
    message(
        "Reading ",
        sample_name,
        ": ",
        basename(current_file)
    )
    
    current_extension <- tolower(
        tools::file_ext(current_file)
    )
    
    if (current_extension == "bed") {
        
        # Standard BED import.
        gr <- rtracklayer::import(
            current_file,
            format = "BED"
        )
        
    } else if (current_extension == "broadpeak") {
        
        # MACS3 broadPeak has:
        #
        # 1  chrom
        # 2  start
        # 3  end
        # 4  name
        # 5  score
        # 6  strand
        # 7  signalValue
        # 8  pValue
        # 9  qValue
        #
        # Columns 7-9 must be declared numeric.
        gr <- rtracklayer::import(
            current_file,
            format = "BED",
            extraCols = c(
                signalValue = "numeric",
                pValue = "numeric",
                qValue = "numeric"
            )
        )
        
    } else {
        
        stop(
            "Unsupported file type: ",
            current_file
        )
    }
    
    if (!inherits(gr, "GRanges")) {
        stop(
            "Imported object is not GRanges for: ",
            current_file
        )
    }
    
    if (length(gr) == 0) {
        stop(
            "No ranges were imported from: ",
            current_file
        )
    }
    
    # Remove unused sequence levels.
    gr <- GenomeInfoDb::keepSeqlevels(
        gr,
        value = GenomeInfoDb::seqlevelsInUse(gr),
        pruning.mode = "coarse"
    )
    
    # Add a unique sample name.
    gr$input_sample <- sample_name
    
    # Add a unique peak ID.
    gr$peak_id <- paste0(
        sample_name,
        "_peak_",
        seq_along(gr)
    )
    
    peak_granges[[sample_name]] <- gr
    
    message(
        "  Imported peaks: ",
        length(gr)
    )
    
    message(
        "  Chromosomes/scaffolds: ",
        length(
            unique(
                as.character(
                    GenomicRanges::seqnames(gr)
                )
            )
        )
    )
}
############################
# 6. Check chromosome matching
############################

message("Checking chromosome/scaffold names...")

for (sample_name in names(peak_granges)) {
    
    peak_seqlevels <- unique(
        as.character(
            GenomicRanges::seqnames(
                peak_granges[[sample_name]]
            )
        )
    )
    
    matching_seqlevels <- intersect(
        peak_seqlevels,
        txdb_seqlevels
    )
    
    message(
        sample_name,
        ": ",
        length(matching_seqlevels),
        " matching sequence names"
    )
    
    if (length(matching_seqlevels) == 0) {
        warning(
            "No chromosome/scaffold names match between ",
            sample_name,
            " and the TxDb.\n",
            "Peak names include: ",
            paste(head(peak_seqlevels, 10), collapse = ", "),
            "\nTxDb names include: ",
            paste(head(txdb_seqlevels, 10), collapse = ", ")
        )
    }
}

############################
# 7. Export normalized BED files
############################

message("Exporting normalized BED files...")

for (sample_name in names(peak_granges)) {
    
    output_bed <- file.path(
        output_dir,
        "normalized_bed",
        paste0(
            sample_name,
            "_normalized.bed"
        )
    )
    
    if (file.exists(output_bed)) {
        file.remove(output_bed)
    }
    
    rtracklayer::export(
        object = peak_granges[[sample_name]],
        con = output_bed,
        format = "BED",
        append = FALSE,
        ignore.strand = TRUE
    )
    
    if (!file.exists(output_bed)) {
        stop(
            "Could not create normalized BED file: ",
            output_bed
        )
    }
}

############################
# 8. Annotate the four files
############################

annotation_objects <- list()
annotation_tables <- list()

for (sample_name in names(peak_granges)) {
    
    message("--------------------------------------------")
    message("Annotating: ", sample_name)
    message("--------------------------------------------")
    
    gr <- peak_granges[[sample_name]]
    
    annotation_result <- ChIPseeker::annotatePeak(
        peak = gr,
        tssRegion = c(-3000, 3000),
        TxDb = txdb,
        level = "transcript",
        assignGenomicAnnotation = TRUE,
        genomicAnnotationPriority = c(
            "Promoter",
            "5UTR",
            "3UTR",
            "Exon",
            "Intron",
            "Downstream",
            "Intergenic"
        ),
        addFlankGeneInfo = TRUE,
        flankDistance = 5000,
        sameStrand = FALSE,
        ignoreOverlap = FALSE,
        overlap = "TSS",
        verbose = TRUE
    )
    
    annotation_objects[[sample_name]] <- annotation_result
    
    annotation_df <- as.data.frame(
        annotation_result
    )
    
    annotation_df$input_sample <- sample_name
    
    annotation_tables[[sample_name]] <- annotation_df
    
    output_table <- file.path(
        output_dir,
        "annotated_tables",
        paste0(
            sample_name,
            "_annotated_peaks.tsv"
        )
    )
    
    write.table(
        annotation_df,
        file = output_table,
        sep = "\t",
        quote = FALSE,
        row.names = FALSE,
        na = ""
    )
    
    message(
        "Annotated peaks written: ",
        output_table
    )
}

# 10. Annotation summaries
############################

summary_list <- list()

for (sample_name in names(annotation_tables)) {
    
    annotation_df <- annotation_tables[[sample_name]]
    
    if (!"annotation" %in% names(annotation_df)) {
        warning(
            "No annotation column found for ",
            sample_name
        )
        next
    }
    
    summary_df <- annotation_df %>%
        dplyr::count(
            annotation,
            sort = TRUE,
            name = "peak_count"
        ) %>%
        dplyr::mutate(
            percentage = 100 * peak_count / sum(peak_count),
            sample = sample_name
        )
    
    summary_list[[sample_name]] <- summary_df
    
    summary_file <- file.path(
        output_dir,
        "annotated_tables",
        paste0(
            sample_name,
            "_annotation_summary.tsv"
        )
    )
    
    write.table(
        summary_df,
        file = summary_file,
        sep = "\t",
        quote = FALSE,
        row.names = FALSE,
        na = ""
    )
}

# 13. Nearest-gene tables
############################

extract_gene_ids <- function(annotation_df) {
    
    candidate_columns <- c(
        "geneId",
        "ENSEMBL",
        "SYMBOL",
        "GENENAME"
    )
    
    candidate_columns <- intersect(
        candidate_columns,
        names(annotation_df)
    )
    
    if (length(candidate_columns) == 0) {
        return(character(0))
    }
    
    gene_values <- unique(
        unlist(
            lapply(
                candidate_columns,
                function(column_name) {
                    as.character(
                        annotation_df[[column_name]]
                    )
                }
            ),
            use.names = FALSE
        )
    )
    
    gene_values <- gene_values[
        !is.na(gene_values) &
            gene_values != "" &
            gene_values != "NA" &
            gene_values != "." &
            gene_values != "0"
    ]
    
    unique(gene_values)
}

gene_summary <- data.frame(
    sample = character(),
    unique_gene_values = integer(),
    stringsAsFactors = FALSE
)

for (sample_name in names(annotation_tables)) {
    
    annotation_df <- annotation_tables[[sample_name]]
    
    gene_values <- extract_gene_ids(
        annotation_df
    )
    
    gene_file <- file.path(
        output_dir,
        "annotated_tables",
        paste0(
            sample_name,
            "_nearest_gene_values.txt"
        )
    )
    
    writeLines(
        gene_values,
        gene_file
    )
    
    gene_summary <- rbind(
        gene_summary,
        data.frame(
            sample = sample_name,
            unique_gene_values = length(gene_values),
            stringsAsFactors = FALSE
        )
    )
}

write.table(
    gene_summary,
    file = file.path(
        output_dir,
        "annotated_tables",
        "All_4_nearest_gene_summary.tsv"
    ),
    sep = "\t",
    quote = FALSE,
    row.names = FALSE
)
