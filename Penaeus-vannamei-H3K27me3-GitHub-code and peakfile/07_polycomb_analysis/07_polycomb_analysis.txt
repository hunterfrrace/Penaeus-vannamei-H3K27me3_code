resolve_path <- function(win_path) {
    if (.Platform$OS.type == "windows") return(win_path)
    wsl_path <- gsub("^D:", "/mnt/d", win_path, ignore.case = TRUE)
    wsl_path <- gsub("\\\\", "/", wsl_path)
    wsl_path
}

# Dynamic File Finder Helper
find_file <- function(pattern, target_dir) {
    found <- list.files(target_dir, pattern = pattern, recursive = TRUE, full.names = TRUE)
    if (length(found) == 0) {
        stop(sprintf("ERROR: File pattern '%s' not found under '%s'", pattern, target_dir))
    }
    found[1] # Return first exact match
}

# Robust BED loader handling non-standard strand markers and column formats
import_bed_robust <- function(file_path) {
    if (!file.exists(file_path)) stop("Missing file: ", file_path)
    df <- fread(file_path, header = FALSE, select = 1:min(6, ncol(fread(file_path, nrows = 5))), fill = TRUE)
    if (nrow(df) == 0) return(GRanges())
    
    colnames(df)[1:min(6, ncol(df))] <- c("chrom", "start", "end", "name", "score", "strand")[1:min(6, ncol(df))]
    
    # 0-based BED to 1-based GRanges conversion
    df$start <- df$start + 1
    
    if ("strand" %in% colnames(df)) {
        df$strand <- as.character(df$strand)
        df$strand[!df$strand %in% c("+", "-")] <- "*"
    } else {
        df$strand <- "*"
    }
    
    GRanges(seqnames = df$chrom, ranges = IRanges(start = df$start, end = df$end), strand = df$strand)
}

import_bed_genes <- function(bed_path, target_gr) {
    peaks <- import_bed_robust(bed_path)
    if (length(peaks) == 0) return(list(n_peaks = 0, genes = character(0)))
    olaps <- findOverlaps(peaks, target_gr)
    hit_ids <- unique(na.omit(target_gr$clean_gene_id[subjectHits(olaps)]))
    list(n_peaks = length(peaks), genes = as.character(hit_ids))
}

# ------------------------------------------------------------------------
# 5. STATISTICAL ENRICHMENT (SHARED BACKGROUND UNIVERSE)
# ------------------------------------------------------------------------
run_enrichment <- function(label, region, fg_genes, gene_set, K_set, N) {
    n <- length(fg_genes)
    k <- length(intersect(fg_genes, gene_set))
    if (n == 0) {
        gene_ratio <- NA; fold <- NA; raw_p <- NA
    } else {
        gene_ratio <- k / n
        bg_ratio   <- K_set / N
        fold       <- gene_ratio / bg_ratio
        raw_p      <- phyper(max(0, k - 1), K_set, N - K_set, n, lower.tail = FALSE)
    }
    data.frame(
        Label = label, Region = region, n = n, k = k, N = N, K = K_set,
        GeneRatio = round(gene_ratio, 4), BgRatio = round(K_set / N, 6),
        Fold = round(fold, 3), Raw_p = raw_p, stringsAsFactors = FALSE
    )
}
