###############################################################################
## 8. Find genomic overlaps
###############################################################################

find_overlap_hits <- function(
        x,
        y,
        min_overlap = 500L
) {
    
    hits <- findOverlaps(
        x,
        y,
        ignore.strand = TRUE
    )
    
    if (length(hits) == 0L) {
        
        return(
            data.table(
                query = integer(0),
                subject = integer(0),
                overlap_bp = integer(0)
            )
        )
    }
    
    overlap_ranges <- pintersect(
        x[
            queryHits(hits)
        ],
        y[
            subjectHits(hits)
        ]
    )
    
    overlap_widths <- width(
        overlap_ranges
    )
    
    keep <- overlap_widths >= min_overlap
    
    data.table(
        
        query = queryHits(hits)[keep],
        
        subject = subjectHits(hits)[keep],
        
        overlap_bp = overlap_widths[keep]
    )
}

## 11. Pairwise consensus region construction
###############################################################################

merge_two_replicates <- function(
        gr1,
        gr2,
        hits,
        sample1,
        sample2
) {
    
    if (nrow(hits) == 0L) {
        
        return(
            GRanges()
        )
    }
    
    output <- vector(
        mode = "list",
        length = nrow(hits)
    )
    
    for (i in seq_len(nrow(hits))) {
        
        q <- gr1[
            hits$query[i]
        ]
        
        s <- gr2[
            hits$subject[i]
        ]
        
        same_chromosome <- as.character(
            seqnames(q)
        ) ==
            as.character(
                seqnames(s)
            )
        
        if (!same_chromosome) {
            next
        }
        
        overlap_start <- max(
            start(q),
            start(s)
        )
        
        overlap_end <- min(
            end(q),
            end(s)
        )
        
        if (overlap_end < overlap_start) {
            next
        }
        
        output[[i]] <- GRanges(
            
            seqnames = seqnames(q),
            
            ranges = IRanges(
                start = overlap_start,
                end = overlap_end
            ),
            
            strand = "*"
        )
    }
    
    output <- output[
        !vapply(
            output,
            is.null,
            logical(1)
        )
    ]
    
    if (length(output) == 0L) {
        return(GRanges())
    }
    
    result <- do.call(
        c,
        output
    )
    
    result <- reduce(
        result,
        ignore.strand = TRUE
    )
    
    names(result) <- paste0(
        sample1,
        "_",
        sample2,
        "_Consensus_",
        seq_along(result)
    )
    
    result
}

## 17. Build exact combination-membership table — corrected
###############################################################################

## Create a non-overlapping reference universe from all extended peaks.

all_extended <- c(
    extended_replicate_granges[["Rep1"]],
    extended_replicate_granges[["Rep2"]],
    extended_replicate_granges[["Rep3"]]
)

union_regions <- reduce(
    all_extended,
    ignore.strand = TRUE
)

if (length(union_regions) == 0L) {
    
    stop(
        "No union regions were created."
    )
}

membership_matrix <- data.table(
    
    Region_ID = paste0(
        "Region_",
        seq_along(
            union_regions
        )
    ),
    
    Chromosome = as.character(
        seqnames(
            union_regions
        )
    ),
    
    Start = start(
        union_regions
    ),
    
    End = end(
        union_regions
    )
)

for (sample_id in sample_levels) {
    
    membership_matrix[
        ,
        (sample_id) := countOverlaps(
            union_regions,
            extended_replicate_granges[[sample_id]],
            ignore.strand = TRUE
        ) > 0L
    ]
}

## Ensure logical columns exist and contain no NA values.

for (sample_id in sample_levels) {
    
    membership_matrix[
        is.na(
            get(sample_id)
        ),
        (sample_id) := FALSE
    ]
}

## Correct combination-label construction.
##
## Do not use:
##   ifelse(condition, "Rep1", NULL)
##
## NULL is length zero and causes the replacement error.

membership_matrix[
    ,
    Combination := vapply(
        seq_len(.N),
        function(i) {
            
            members <- character(0)
            
            if (isTRUE(Rep1[i])) {
                members <- c(
                    members,
                    "Rep1"
                )
            }
            
            if (isTRUE(Rep2[i])) {
                members <- c(
                    members,
                    "Rep2"
                )
            }
            
            if (isTRUE(Rep3[i])) {
                members <- c(
                    members,
                    "Rep3"
                )
            }
            
            if (length(members) == 0L) {
                
                return(
                    "None"
                )
            }
            
            paste(
                members,
                collapse = "+"
            )
        },
        character(1)
    )
]

## Define the number of replicates supporting each union region.

membership_matrix[
    ,
    Number_of_replicates := rowSums(
        .SD
    ),
    .SDcols = sample_levels
]

## Order columns for easier inspection.

setcolorder(
    membership_matrix,
    c(
        "Region_ID",
        "Chromosome",
        "Start",
        "End",
        "Rep1",
        "Rep2",
        "Rep3",
        "Number_of_replicates",
        "Combination"
    )
)

## Save region-level membership.
